// Power BI DAX measures — Lending Club credit-risk dashboard
// Import credit_risk/output/powerbi_scored_loan_data.csv as table 'powerbi_scored_loan_data'

// =============================================================================
// ORIGINAL MEASURES
// =============================================================================

// 1. Expected Credit Loss (ECL) based on IFRS 9 framework (PD x LGD x EAD)
Expected Loss =
SUMX(
    'powerbi_scored_loan_data',
    'powerbi_scored_loan_data'[loan_amnt]
        * 'powerbi_scored_loan_data'[probability_of_default]
        * 0.60
)

// 2. Average Debt-to-Income
Avg DTI = AVERAGE('powerbi_scored_loan_data'[dti])

// 3. Dynamic Misclassification Rate Tracker
Net Misclassification Rate =
VAR TotalActualDefault =
    CALCULATE(
        COUNT('powerbi_scored_loan_data'[default_flag]),
        'powerbi_scored_loan_data'[default_flag] = 1
    )
VAR TotalActualSafe =
    CALCULATE(
        COUNT('powerbi_scored_loan_data'[default_flag]),
        'powerbi_scored_loan_data'[default_flag] = 0
    )
RETURN
    (TotalActualDefault * 0.18 + TotalActualSafe * 0.12)
        / COUNT('powerbi_scored_loan_data'[default_flag])

// =============================================================================
// TIER A — PORTFOLIO RISK KPIs
// =============================================================================

// Exposure-weighted portfolio PD
Portfolio Weighted PD =
DIVIDE(
    SUMX(
        'powerbi_scored_loan_data',
        'powerbi_scored_loan_data'[loan_amnt]
            * 'powerbi_scored_loan_data'[probability_of_default]
    ),
    SUM('powerbi_scored_loan_data'[loan_amnt])
)

// ECL as % of book (reserve intensity)
ECL Coverage Ratio =
DIVIDE(
    [Expected Loss],
    SUM('powerbi_scored_loan_data'[loan_amnt])
)

// Observed (actual) default rate for model calibration comparison
Observed Default Rate =
AVERAGE('powerbi_scored_loan_data'[default_flag])

// Use with vintage_year on rows/columns for cohort chart
Vintage Default Rate =
AVERAGE('powerbi_scored_loan_data'[default_flag])

// Grade exposure share — use grade on rows; this is portfolio total for denominator
Total Exposure =
SUM('powerbi_scored_loan_data'[loan_amnt])

Grade Exposure Share =
DIVIDE(
    SUM('powerbi_scored_loan_data'[loan_amnt]),
    CALCULATE(SUM('powerbi_scored_loan_data'[loan_amnt]), ALL('powerbi_scored_loan_data'[grade]))
)

// =============================================================================
// TIER B — PRICING, SCORECARD, STRESS
// =============================================================================

// Share of book flagged as underpriced (high PD + negative pricing adequacy)
Underpriced Exposure Pct =
DIVIDE(
    CALCULATE(
        SUM('powerbi_scored_loan_data'[loan_amnt]),
        'powerbi_scored_loan_data'[underpriced_flag] = 1
    ),
    SUM('powerbi_scored_loan_data'[loan_amnt])
)

Avg Pricing Adequacy =
AVERAGE('powerbi_scored_loan_data'[pricing_adequacy])

Avg Risk Spread =
AVERAGE('powerbi_scored_loan_data'[risk_spread])

// Severe stress ECL (PD x 2.0) — illustrative scenario for dashboard KPI card
Stressed ECL Severe =
SUMX(
    'powerbi_scored_loan_data',
    'powerbi_scored_loan_data'[loan_amnt]
        * MIN('powerbi_scored_loan_data'[probability_of_default] * 2.0, 0.99)
        * 0.60
)

// Application score band default rate — put application_score on a visual axis
// (or create a calculated column for score decile in Power Query)
Application Score Default Rate =
AVERAGE('powerbi_scored_loan_data'[default_flag])

// Optional: if you also load stress_test_results.csv as 'stress_test_results'
Severe ECL From Table =
CALCULATE(
    SUM('stress_test_results'[total_ecl]),
    'stress_test_results'[scenario] = "severe"
)
